   318→async function seedHandbook() {
   319→  console.log("=== Seeding GED Handbook Data ===\n");
   320→
   321→  // Clear existing handbook data
   322→  await prisma.handbookContent.deleteMany();
   323→  await prisma.handbookTopic.deleteMany();
   324→
   325→  let totalTopics = 0;
   326→  let totalContents = 0;
   327→
   328→  // Map to track topic IDs for question linking
   329→  const topicIdMap: Record<string, string> = {};
   330→
   331→  for (const topicData of HANDBOOK_DATA) {
   332→    const subject = await prisma.subject.findUnique({
   333→      where: { code: topicData.subjectCode },
   334→    });
   335→
   336→    if (!subject) {
   337→      console.warn(`  Subject ${topicData.subjectCode} not found, skipping`);
   338→      continue;
   339→    }
   340→
   341→    const topic = await prisma.handbookTopic.create({
   342→      data: {
   343→        subjectId: subject.id,
   344→        title: topicData.title,
   345→        titleTh: topicData.titleTh,
   346→        titleMm: topicData.titleMm,
   347→        categoryType: topicData.categoryType,
   348→        sortOrder: topicData.sortOrder,
   349→      },
   350→    });
   351→
   352→    // Store topic ID for linking (key: subjectCode + first word of title)
   353→    const key = `${topicData.subjectCode}_${topicData.categoryType}_${topicData.sortOrder}`;
   354→    topicIdMap[key] = topic.id;
   355→
   356→    for (const contentData of topicData.contents) {
   357→      await prisma.handbookContent.create({
   358→        data: {
   359→          topicId: topic.id,
   360→          contentBodyEn: contentData.contentBodyEn,
   361→          contentBodyTh: contentData.contentBodyTh,
   362→          contentBodyMm: contentData.contentBodyMm,
   363→          keyTakeaways: JSON.stringify(contentData.keyTakeaways),
   364→          formulaOrRules: JSON.stringify(contentData.formulaOrRules),
   365→          sortOrder: contentData.sortOrder,
   366→        },
   367→      });
   368→      totalContents++;
   369→    }
   370→
   371→    totalTopics++;
   372→    console.log(`  [${topicData.subjectCode}] ${topicData.categoryType}: ${topicData.title}`);
   373→  }
   374→
   375→  console.log(`\nTotal handbook topics: ${totalTopics}`);
   376→  console.log(`Total handbook contents: ${totalContents}`);
   377→
   378→  // =======================================================================
   379→  // LINK EXISTING QUESTIONS TO HANDBOOK TOPICS
   380→  // =======================================================================
   381→  console.log("\n--- Linking Questions to Handbook Topics ---");
   382→
   383→  // Get all subjects with their handbook topics
   384→  const subjects = await prisma.subject.findMany({
   385→    include: {
   386→      handbookTopics: true,
   387→      questions: { where: { relatedConceptId: null } },
   388→    },
   389→  });
   390→
   391→  let linkedCount = 0;
   392→
   393→  for (const subject of subjects) {
   394→    const textbookTopics = subject.handbookTopics.filter(t => t.categoryType === "textbook");
   395→    if (textbookTopics.length === 0) continue;
   396→
   397→    const questionsToLink = subject.questions;
   398→    if (questionsToLink.length === 0) continue;
   399→
   400→    // Distribute questions across textbook topics evenly
   401→    for (let i = 0; i < questionsToLink.length; i++) {
   402→      const topicIndex = i % textbookTopics.length;
   403→      const topicId = textbookTopics[topicIndex].id;
   404→
   405→      await prisma.question.update({
   406→        where: { id: questionsToLink[i].id },
   407→        data: { relatedConceptId: topicId },
   408→      });
   409→      linkedCount++;
   410→    }
   411→
   412→    console.log(`  ${subject.code}: linked ${questionsToLink.length} questions to ${textbookTopics.length} handbook topics`);
   413→  }
   414→
   415→  console.log(`Total questions linked: ${linkedCount}`);
   416→}
   417→
   418→seedHandbook()

   420→    console.log("\n=== Handbook Seed Complete ===");
   421→    await prisma.$disconnect();

   424→    console.error("Handbook seed failed:", e);
   425→    await prisma.$disconnect();
   426→    process.exit(1);
   427→  });
