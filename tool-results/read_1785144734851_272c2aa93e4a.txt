   394→async function main() {
   395→  for (const [_, subjectData] of Object.entries(DATA)) {
   396→    const subject = await p.subject.findFirst({ where: { code: subjectData.code } });
   397→    if (!subject) { console.log(`SKIP ${subjectData.code}`); continue; }
   398→
   399→    for (const [lessonTitle, questions] of Object.entries(subjectData.lessons)) {
   400→      const lesson = await p.lesson.findFirst({
   401→        where: { title: lessonTitle, topic: { module: { subjectId: subject.id } } },
   402→      });
   403→      if (!lesson) { console.log(`  SKIP lesson: ${lessonTitle}`); continue; }
   404→
   405→      const existing = await p.question.findMany({ where: { lessonId: lesson.id }, include: { answers: true }, orderBy: { createdAt: "asc" } });
   406→      console.log(`  ${subjectData.code} | ${lessonTitle} | existing: ${existing.length} | new: ${questions.length}`);
   407→
   408→      // อัปเดตข้อเดิม
   409→      const updateCount = Math.min(existing.length, questions.length);
   410→      for (let i = 0; i < updateCount; i++) {
   411→        const q = questions[i], eq = existing[i];
   412→        await p.question.update({ where: { id: eq.id }, data: { questionText: q.questionText, explanation: q.explanation, difficulty: q.difficulty } });
   413→        for (let j = 0; j < eq.answers.length && j < q.answers.length; j++) {
   414→          await p.answer.update({ where: { id: eq.answers[j].id }, data: { content: q.answers[j].content, isCorrect: q.answers[j].isCorrect, explanation: q.answers[j].explanation || null } });
   415→        }
   416→      }
   417→      // เพิ่มใหม่
   418→      for (let i = updateCount; i < questions.length; i++) {
   419→        const q = questions[i];
   420→        await p.question.create({ data: { subjectId: subject.id, lessonId: lesson.id, questionType: "multiple_choice", difficulty: q.difficulty, questionText: q.questionText, explanation: q.explanation, points: 1, isActive: true, answers: { create: q.answers.map((a, idx) => ({ content: a.content, isCorrect: a.isCorrect, sortOrder: idx, explanation: a.explanation || null })) } } });
   421→      }
   422→    }
   423→  }
   424→  const total = await p.question.count({ where: { questionText: { not: null } } });
   425→  console.log(`\nDone! Questions with text: ${total}/${await p.question.count()}`);
   426→}
   427→main().catch(console.error).finally(() => p.$disconnect());
