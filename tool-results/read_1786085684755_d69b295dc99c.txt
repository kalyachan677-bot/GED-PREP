   225→   225→  let totalAdded = 0;

   230→   230→    const lessonTitle = lesson.title;
   231→   231→    const questionsToAdd = QUESTIONS[lessonTitle];
   232→   232→
   233→   233→    if (!questionsToAdd || questionsToAdd.length === 0) {
   234→   234→      console.log(`SKIP: ${subjectCode} > ${lessonTitle} (no new questions)`);
   235→   235→      continue;
   236→   236→    }
   237→   237→
   238→   238→    for (const [questionText, answers, difficulty, explanation, tags] of questionsToAdd) {
   239→   239→      try {
   240→   240→        await prisma.question.create({
   241→   241→          data: {
   242→   242→            id: uuid(),

   244→   244→            difficulty,
   245→   245→            questionText,
   246→   246→            explanation,
   247→   247→            hintText: "",
   248→   248→            tags: JSON.stringify(tags),
   249→   249→            isActive: true,
   250→   250→            points: difficulty === "hard" ? 3 : difficulty === "medium" ? 2 : 1,
   251→   251→            subjectId: lesson.topic.module.subject.id,
   252→   252→            lessonId: lesson.id,
   253→   253→            answers: {
   254→   254→              create: answers.map((a, i) => ({
   255→   255→                id: uuid(),
   256→   256→                content: a[0],
   257→   257→                isCorrect: a[1],
   258→   258→                sortOrder: i,
   259→   259→              }))
   260→   260→            }
   261→   261→          }
   262→   262→        });
   263→   263→        totalAdded++;
   264→   264→      } catch (e: any) {
   265→   265→        errors.push(`${subjectCode} > ${lessonTitle}: ${e.message}`);
   266→   266→      }
   267→   267→    }
   268→   268→    console.log(`OK: ${subjectCode} > ${lessonTitle}: +${questionsToAdd.length} questions`);
   269→   269→  }

   275→   275→  }

   277→   277→  // Print summary
